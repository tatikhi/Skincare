  <script>
    // --- APP STATE & SCHEDULE CONFIGURATION ---
    const STORAGE_KEY = 'skincare_tracker_state_v1';
    let currentFilter = 'all';

    // 28-Day Schedule Structure Starting on Wednesday
    // Retinol Schedule: 
    // Weeks 1-2: Wednesday (Day 1, Day 8) and Saturday (Day 4, Day 11)
    // Weeks 3-4: Wednesday (Day 15, 22), Saturday (Day 18, 25), Tuesday (Day 21, 27)
    const daysOfWeek = ['Среда', 'Четверг', 'Пятница', 'Суббота', 'Воскресенье', 'Понедельник', 'Вторник'];

    const retinolDaysSet = new Set([1, 4, 8, 11, 15, 18, 21, 24, 27]);

    function isRetinolDay(dayNum) {
      return retinolDaysSet.has(dayNum);
    }

    // Generate Full 28 Days Dataset
    function generateDaysData() {
      const days = [];
      for (let i = 1; i <= 28; i++) {
        const weekNum = Math.ceil(i / 7);
        const dayOfWeek = daysOfWeek[(i - 1) % 7];
        const isRetinol = isRetinolDay(i);

        days.push({
          id: `day_${i}`,
          dayNumber: i,
          week: weekNum,
          dayOfWeek: dayOfWeek,
          isRetinol: isRetinol,
          morningSteps: [
            { id: `day_${i}_m1`, text: 'Мусс с алоэ вера', note: 'Очищение' },
            { id: `day_${i}_m2`, text: 'Крем NMF-Protection', note: 'Восстановление и увлажнение' },
            { id: `day_${i}_m3`, text: 'SPF-крем', note: 'Защита от солнца' }
          ],
          eveningSteps: isRetinol ? [
            { id: `day_${i}_e1`, text: 'Мусс с алоэ вера', note: 'Очищение' },
            { id: `day_${i}_e2`, text: 'Retiderm 0.25', note: 'Нанести на сухую кожу (через 15 мин после умывания)', isRetinolBadge: true },
            { id: `day_${i}_e3`, text: 'Крем NMF-Protection', note: 'Закрыть через 20 минут после ретинола' }
          ] : [
            { id: `day_${i}_e1`, text: 'Мусс с алоэ вера', note: 'Очищение' },
            { id: `day_${i}_e2`, text: 'Крем NMF-Protection', note: 'Восстановление и увлажнение' }
          ]
        });
      }
      return days;
    }

    const scheduleData = generateDaysData();

    // LocalStorage Management
    function loadSavedState() {
      try {
        const saved = localStorage.getItem(STORAGE_KEY);
        return saved ? JSON.parse(saved) : {};
      } catch (e) {
        console.error("Failed to load state", e);
        return {};
      }
    }

    function saveState(state) {
      try {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
      } catch (e) {
        console.error("Failed to save state", e);
      }
    }

    let userState = loadSavedState();

    // --- UI RENDER LOGIC ---
    function renderApp() {
      const container = document.getElementById('daysContainer');
      container.innerHTML = '';

      const filteredDays = scheduleData.filter(d => {
        if (currentFilter === 'all') return true;
        return d.week === parseInt(currentFilter);
      });

      filteredDays.forEach(day => {
        const card = createDayCard(day);
        container.appendChild(card);
      });

      updateOverallProgress();
    }

    function createDayCard(day) {
      const card = document.createElement('div');
      
      // Calculate completion for this specific day
      const allStepIds = [...day.morningSteps.map(s => s.id), ...day.eveningSteps.map(s => s.id)];
      const completedSteps = allStepIds.filter(id => !!userState[id]);
      const dayProgressPct = Math.round((completedSteps.length / allStepIds.length) * 100);
      const isDayComplete = completedSteps.length === allStepIds.length;

      card.className = `bg-white rounded-2xl p-5 shadow-sm border transition-all duration-300 relative ${
        isDayComplete 
          ? 'border-emerald-300 bg-emerald-50/10' 
          : day.isRetinol 
            ? 'border-amber-200/80 hover:border-amber-300' 
            : 'border-brand-rose/30 hover:border-brand-sage/40'
      }`;

      card.innerHTML = `
        <!-- Card Header -->
        <div class="flex items-center justify-between border-b border-gray-100 pb-3 mb-4">
          <div class="flex items-center gap-3">
            <span class="w-9 h-9 rounded-xl ${day.isRetinol ? 'bg-amber-100 text-amber-800' : 'bg-brand-sageLight text-brand-sage'} font-serif font-bold text-sm flex items-center justify-center">
              ${day.dayNumber}
            </span>
            <div>
              <div class="flex items-center gap-2">
                <h3 class="font-semibold text-sm text-brand-dark">${day.dayOfWeek}</h3>
                <span class="text-[10px] text-brand-muted bg-gray-100 px-2 py-0.5 rounded-full font-medium">Неделя ${day.week}</span>
              </div>
              <p class="text-xs text-brand-muted mt-0.5">
                ${day.isRetinol ? '<span class="text-amber-700 font-medium"><i class="fa-solid fa-sparkles text-amber-500"></i> День с Ретинолом</span>' : 'Базовый восстановливающий уход'}
              </p>
            </div>
          </div>

          <!-- Day Completion Badge -->
          <div class="text-right">
            <span id="pct_${day.id}" class="text-xs font-bold ${isDayComplete ? 'text-emerald-600' : 'text-brand-muted'}">
              ${dayProgressPct}%
            </span>
            <div class="w-12 bg-gray-100 rounded-full h-1.5 mt-1 overflow-hidden">
              <div id="bar_${day.id}" class="progress-bar-fill ${isDayComplete ? 'bg-emerald-500' : 'bg-brand-sage'} h-full" style="width: ${dayProgressPct}%"></div>
            </div>
          </div>
        </div>

        <!-- Routine Slots Grid -->
        <div class="space-y-4">
          
          <!-- MORNING (10:00) -->
          <div class="bg-amber-50/40 rounded-xl p-3.5 border border-amber-100/60">
            <div class="flex items-center justify-between mb-2.5">
              <span class="text-xs font-semibold text-amber-900 flex items-center gap-1.5">
                <i class="fa-regular fa-sun text-amber-500"></i> Утро (10:00)
              </span>
              <span class="text-[10px] text-amber-700/70 font-medium">3 шага</span>
            </div>
            <div class="space-y-2">
              ${day.morningSteps.map(step => renderStepRow(step)).join('')}
            </div>
          </div>

          <!-- EVENING (21:00) -->
          <div class="${day.isRetinol ? 'bg-amber-50/60 border-amber-200/80' : 'bg-indigo-50/30 border-indigo-100/60'} rounded-xl p-3.5 border">
            <div class="flex items-center justify-between mb-2.5">
              <span class="text-xs font-semibold ${day.isRetinol ? 'text-amber-900' : 'text-indigo-900'} flex items-center gap-1.5">
                <i class="fa-regular fa-moon ${day.isRetinol ? 'text-amber-600' : 'text-indigo-400'}"></i> Вечер (21:00)
              </span>
              <span class="text-[10px] ${day.isRetinol ? 'text-amber-800 bg-amber-200/60' : 'text-indigo-700 bg-indigo-100/60'} px-2 py-0.5 rounded-full font-medium">
                ${day.isRetinol ? 'Retiderm 0.25%' : 'Базовое увлажнение'}
              </span>
            </div>
            <div class="space-y-2">
              ${day.eveningSteps.map(step => renderStepRow(step)).join('')}
            </div>
          </div>

        </div>
      `;

      return card;
    }

    function renderStepRow(step) {
      const isChecked = !!userState[step.id];
      return `
        <label class="flex items-start gap-2.5 cursor-pointer group select-none">
          <input 
            type="checkbox" 
            class="custom-checkbox mt-0.5" 
            id="${step.id}" 
            ${isChecked ? 'checked' : ''} 
            onchange="toggleStep('${step.id}')"
          />
          <div class="text-xs leading-tight">
            <span class="font-medium text-brand-dark group-hover:text-black ${isChecked ? 'line-through text-gray-400' : ''}">
              ${step.text}
            </span>
            ${step.note ? `<p class="text-[11px] text-brand-muted mt-0.5 ${isChecked ? 'text-gray-300' : ''}">${step.note}</p>` : ''}
          </div>
        </label>
      `;
    }

    // --- INTERACTION & PROGRESS LOGIC ---
    function toggleStep(stepId) {
      const checkbox = document.getElementById(stepId);
      if (!checkbox) return;

      userState[stepId] = checkbox.checked;
      saveState(userState);

      // Find affected day
      const affectedDay = scheduleData.find(d => 
        d.morningSteps.some(s => s.id === stepId) || d.eveningSteps.some(s => s.id === stepId)
      );

      if (affectedDay) {
        updateDayProgress(affectedDay);
      }

      updateOverallProgress();
    }

    function updateDayProgress(day) {
      const allStepIds = [...day.morningSteps.map(s => s.id), ...day.eveningSteps.map(s => s.id)];
      const completedSteps = allStepIds.filter(id => !!userState[id]);
      const pct = Math.round((completedSteps.length / allStepIds.length) * 100);

      const pctElem = document.getElementById(`pct_${day.id}`);
      const barElem = document.getElementById(`bar_${day.id}`);

      if (pctElem) {
        pctElem.innerText = `${pct}%`;
        pctElem.className = `text-xs font-bold ${pct === 100 ? 'text-emerald-600' : 'text-brand-muted'}`;
      }

      if (barElem) {
        barElem.style.width = `${pct}%`;
        barElem.className = `progress-bar-fill ${pct === 100 ? 'bg-emerald-500' : 'bg-brand-sage'} h-full`;
      }
    }

    function updateOverallProgress() {
      let totalSteps = 0;
      let completedSteps = 0;

      scheduleData.forEach(day => {
        const stepIds = [...day.morningSteps.map(s => s.id), ...day.eveningSteps.map(s => s.id)];
        totalSteps += stepIds.length;
        stepIds.forEach(id => {
          if (userState[id]) completedSteps++;
        });
      });

      const overallPct = totalSteps > 0 ? Math.round((completedSteps / totalSteps) * 100) : 0;

      document.getElementById('overallPercentage').innerText = `${overallPct}%`;
      document.getElementById('overallFraction').innerText = `(${completedSteps} из ${totalSteps} шагов выполнено)`;
      document.getElementById('overallProgressBar').style.width = `${overallPct}%`;
    }

    // --- FILTERS ---
    function setFilter(filter) {
      currentFilter = filter;

      document.querySelectorAll('.filter-btn').forEach(btn => {
        btn.className = 'filter-btn px-3.5 py-1.5 text-xs sm:text-sm font-medium rounded-xl transition-all border border-gray-200 bg-white text-brand-muted hover:border-brand-sage';
      });

      const activeBtn = document.getElementById(`btn-filter-${filter}`);
      if (activeBtn) {
        activeBtn.className = 'filter-btn px-3.5 py-1.5 text-xs sm:text-sm font-medium rounded-xl transition-all border border-brand-sage bg-brand-sage text-white shadow-sm';
      }

      renderApp();
    }

    // --- RESET MODAL LOGIC ---
    function confirmReset() {
      document.getElementById('resetModal').classList.remove('hidden');
    }

    function closeResetModal() {
      document.getElementById('resetModal').classList.add('hidden');
    }

    function executeReset() {
      userState = {};
      saveState(userState);
      closeResetModal();
      renderApp();
    }

    // Initialize App
    window.onload = function() {
      renderApp();
    };
  </script>
