    <script>
        const daysOfWeek = ['Среда', 'Четверг', 'Пятница', 'Суббота', 'Воскресенье', 'Понедельник', 'Вторник'];

        function isRetinolDay(week, dayIndex) {
            // Неделя 3: Среда(0) и Суббота(3)
            if (week === 3 && (dayIndex === 0 || dayIndex === 3)) return true;
            // Неделя 4: Среда(0), Пятница(2), Воскресенье(4)
            if (week === 4 && (dayIndex === 0 || dayIndex === 2 || dayIndex === 4)) return true;
            return false;
        }

        function loadState() {
            const saved = localStorage.getItem('skincare_tracker_app');
            return saved ? JSON.parse(saved) : {};
        }

        function saveState(state) {
            localStorage.setItem('skincare_tracker_app', JSON.stringify(state));
            updateOverallProgress();
        }

        function toggleTask(id) {
            const state = loadState();
            state[id] = !state[id];
            saveState(state);
        }

        function updateOverallProgress() {
            const state = loadState();
            const checkboxes = document.querySelectorAll('input[type="checkbox"]');
            if (checkboxes.length === 0) return;

            let checked = 0;
            checkboxes.forEach(cb => {
                if (state[cb.id]) checked++;
            });

            const percent = Math.round((checked / checkboxes.length) * 100);
            document.getElementById('overall-progress-text').innerText = percent + '%';
        }

        function renderCalendar() {
            const container = document.getElementById('calendar-content');
            container.innerHTML = '';
            const state = loadState();

            let dayCounter = 1;

            for (let week = 1; week <= 4; week++) {
                const weekWrapper = document.createElement('div');
                weekWrapper.className = `week-block week-${week}`;

                let badgeText = week === 1 ? 'Адаптация (без ретинола)' : week === 2 ? 'Базовый уход' : week === 3 ? 'Ретинол 2 р/нед' : 'Ретинол 3 р/нед';
                let isRetinolWeek = week >= 3;

                weekWrapper.innerHTML = `
                    <div class="week-title">
                        Неделя ${week} 
                        <span class="badge ${isRetinolWeek ? 'retinol' : ''}">${badgeText}</span>
                    </div>
                `;

                const grid = document.createElement('div');
                grid.className = 'days-grid';

                for (let dayIdx = 0; dayIdx < 7; dayIdx++) {
                    const dayNum = dayCounter;
                    const dayName = daysOfWeek[dayIdx];
                    const retinol = isRetinolDay(week, dayIdx);

                    const card = document.createElement('div');
                    card.className = 'day-card';

                    const m1Key = `d${dayNum}_m1`;
                    const m2Key = `d${dayNum}_m2`;
                    const m3Key = `d${dayNum}_m3`;

                    const e1Key = `d${dayNum}_e1`;
                    const e2Key = `d${dayNum}_e2`;
                    const e3Key = `d${dayNum}_e3`;

                    let eveningHtml = '';
                    if (retinol) {
                        eveningHtml = `
                            <label class="task-item">
                                <input type="checkbox" id="${e1Key}" ${state[e1Key] ? 'checked' : ''} onchange="toggleTask('${e1Key}')">
                                <span>Умывание муссом с алоэ</span>
                            </label>
                            <label class="task-item">
                                <input type="checkbox" id="${e2Key}" ${state[e2Key] ? 'checked' : ''} onchange="toggleTask('${e2Key}')">
                                <span class="retinol-text">Сухая кожа (10 мин) + Retiderm 0.25</span>
                            </label>
                            <label class="task-item">
                                <input type="checkbox" id="${e3Key}" ${state[e3Key] ? 'checked' : ''} onchange="toggleTask('${e3Key}')">
                                <span>Пауза 20 мин + крем NMF-Protection</span>
                            </label>
                        `;
                    } else {
                        eveningHtml = `
                            <label class="task-item">
                                <input type="checkbox" id="${e1Key}" ${state[e1Key] ? 'checked' : ''} onchange="toggleTask('${e1Key}')">
                                <span>Умывание муссом с алоэ</span>
                            </label>
                            <label class="task-item">
                                <input type="checkbox" id="${e2Key}" ${state[e2Key] ? 'checked' : ''} onchange="toggleTask('${e2Key}')">
                                <span>Крем NMF-Protection</span>
                            </label>
                        `;
                    }

                    card.innerHTML = `
                        <div>
                            <div class="day-header">
                                <span class="day-name">${dayName}</span>
                                <span class="day-num">День ${dayNum}</span>
                            </div>

                            <div class="section-box">
                                <span class="section-time">☀️ Утро (10:00)</span>
                                <label class="task-item">
                                    <input type="checkbox" id="${m1Key}" ${state[m1Key] ? 'checked' : ''} onchange="toggleTask('${m1Key}')">
                                    <span>Умывание муссом с алоэ</span>
                                </label>
                                <label class="task-item">
                                    <input type="checkbox" id="${m2Key}" ${state[m2Key] ? 'checked' : ''} onchange="toggleTask('${m2Key}')">
                                    <span>Крем NMF-Protection</span>
                                </label>
                                <label class="task-item">
                                    <input type="checkbox" id="${m3Key}" ${state[m3Key] ? 'checked' : ''} onchange="toggleTask('${m3Key}')">
                                    <span>Солнцезащитный крем (SPF)</span>
                                </label>
                            </div>

                            <div class="section-box">
                                <span class="section-time">🌙 Вечер (21:00)</span>
                                ${eveningHtml}
                            </div>
                        </div>
                    `;

                    grid.appendChild(card);
                    dayCounter++;
                }

                weekWrapper.appendChild(grid);
                container.appendChild(weekWrapper);
            }

            updateOverallProgress();
        }

        function filterWeek(weekNum) {
            const buttons = document.querySelectorAll('.filter-buttons .btn');
            buttons.forEach(btn => btn.classList.remove('active'));
            event.target.classList.add('active');

            const blocks = document.querySelectorAll('.week-block');
            blocks.forEach(block => {
                if (weekNum === 'all') {
                    block.style.display = 'block';
                } else {
                    if (block.classList.contains(`week-${weekNum}`)) {
                        block.style.display = 'block';
                    } else {
                        block.style.display = 'none';
                    }
                }
            });
        }

        function resetAll() {
            if (confirm('Сбросить весь отмеченный прогресс?')) {
                localStorage.removeItem('skincare_tracker_app');
                renderCalendar();
            }
        }

        document.addEventListener('DOMContentLoaded', renderCalendar);
    </script>
