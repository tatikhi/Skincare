    <script>
        var days = ['Среда', 'Четверг', 'Пятница', 'Суббота', 'Воскресенье', 'Понедельник', 'Вторник'];

        function getData() {
            var data = localStorage.getItem('skin_tracker');
            return data ? JSON.parse(data) : {};
        }

        function saveData(data) {
            localStorage.setItem('skin_tracker', JSON.stringify(data));
            calcProgress();
        }

        function checkTask(id) {
            var data = getData();
            data[id] = !data[id];
            saveData(data);
        }

        function calcProgress() {
            var data = getData();
            var boxes = document.querySelectorAll('input[type="checkbox"]');
            if (boxes.length === 0) return;
            
            var count = 0;
            for (var i = 0; i < boxes.length; i++) {
                if (data[boxes[i].id]) {
                    count++;
                }
            }
            var percent = Math.round((count / boxes.length) * 100);
            document.getElementById('progress').innerText = percent + '%';
        }

        function init() {
            var container = document.getElementById('calendar');
            var data = getData();
            var dayNum = 1;

            var html = '';

            for (var w = 1; w <= 4; w++) {
                html += '<div class="week-block week-' + w + '">';
                html += '<div class="week-title">Неделя ' + w + '</div>';
                html += '<div class="grid">';

                for (var d = 0; d < 7; d++) {
                    var isRet = (w === 3 && (d === 0 || d === 3)) || (w === 4 && (d === 0 || d === 2 || d === 4));
                    
                    html += '<div class="card ' + (isRet ? 'retinol-card' : '') + '">';
                    html += '<div class="card-header"><span>' + days[d] + '</span><span>День ' + dayNum + '</span></div>';
                    
                    // Утро
                    html += '<div class="time-title">☀️ Утро (10:00)</div>';
                    html += '<label class="task"><input type="checkbox" id="d' + dayNum + '_m1" ' + (data['d' + dayNum + '_m1'] ? 'checked' : '') + ' onchange="checkTask(\'d' + dayNum + '_m1\')"> Умывание муссом с алоэ</label>';
                    html += '<label class="task"><input type="checkbox" id="d' + dayNum + '_m2" ' + (data['d' + dayNum + '_m2'] ? 'checked' : '') + ' onchange="checkTask(\'d' + dayNum + '_m2\')"> Крем NMF-Protection</label>';
                    html += '<label class="task"><input type="checkbox" id="d' + dayNum + '_m3" ' + (data['d' + dayNum + '_m3'] ? 'checked' : '') + ' onchange="checkTask(\'d' + dayNum + '_m3\')"> Солнцезащитный крем (SPF)</label>';

                    // Вечер
                    html += '<div class="time-title">🌙 Вечер (21:00)</div>';
                    if (isRet) {
                        html += '<label class="task"><input type="checkbox" id="d' + dayNum + '_e1" ' + (data['d' + dayNum + '_e1'] ? 'checked' : '') + ' onchange="checkTask(\'d' + dayNum + '_e1\')"> Умывание муссом с алоэ</label>';
                        html += '<label class="task"><input type="checkbox" id="d' + dayNum + '_e2" ' + (data['d' + dayNum + '_e2'] ? 'checked' : '') + ' onchange="checkTask(\'d' + dayNum + '_e2\')"> <span class="retinol-text">Сухая кожа (10 мин) + Retiderm 0.25</span></label>';
                        html += '<label class="task"><input type="checkbox" id="d' + dayNum + '_e3" ' + (data['d' + dayNum + '_e3'] ? 'checked' : '') + ' onchange="checkTask(\'d' + dayNum + '_e3\')"> Пауза 20 мин + крем NMF-Protection</label>';
                    } else {
                        html += '<label class="task"><input type="checkbox" id="d' + dayNum + '_e1" ' + (data['d' + dayNum + '_e1'] ? 'checked' : '') + ' onchange="checkTask(\'d' + dayNum + '_e1\')"> Умывание муссом с алоэ</label>';
                        html += '<label class="task"><input type="checkbox" id="d' + dayNum + '_e2" ' + (data['d' + dayNum + '_e2'] ? 'checked' : '') + ' onchange="checkTask(\'d' + dayNum + '_e2\')"> Крем NMF-Protection</label>';
                    }

                    html += '</div>';
                    dayNum++;
                }

                html += '</div></div>';
            }

            container.innerHTML = html;
            calcProgress();
        }

        function showWeek(w, btn) {
            var tabs = document.querySelectorAll('.tab');
            for (var i = 0; i < tabs.length; i++) tabs[i].classList.remove('active');
            btn.classList.add('active');

            var blocks = document.querySelectorAll('.week-block');
            for (var j = 0; j < blocks.length; j++) {
                if (w === 'all') {
                    blocks[j].style.display = 'block';
                } else {
                    blocks[j].style.display = blocks[j].classList.contains('week-' + w) ? 'block' : 'none';
                }
            }
        }

        function clearData() {
            if (confirm('Сбросить весь прогресс?')) {
                localStorage.removeItem('skin_tracker');
                init();
            }
        }

        window.onload = init;
