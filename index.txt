    <script>
        var daysOfWeek = ['Среда', 'Четверг', 'Пятница', 'Суббота', 'Воскресенье', 'Понедельник', 'Вторник'];

        function isRetinolDay(week, dayIndex) {
            // Неделя 3: Среда(0), Суббота(3)
            if (week === 3 && (dayIndex === 0 || dayIndex === 3)) return true;
            // Неделя 4: Среда(0), Пятница(2), Воскресенье(4)
            if (week === 4 && (dayIndex === 0 || dayIndex === 2 || dayIndex === 4)) return true;
            return false;
        }

        function getSavedState() {
            try {
                var saved = localStorage.getItem('skincare_gh_pages_data');
                return saved ? JSON.parse(saved) : {};
            } catch(e) {
                return {};
            }
        }

        function saveState(state) {
            try {
                localStorage.setItem('skincare_gh_pages_data', JSON.stringify(state));
            } catch(e) {}
            updateProgress();
        }

        function toggleTask(id) {
            var state = getSavedState();
            state[id] = !state[id];
            saveState(state);
        }

        function updateProgress() {
            var state = getSavedState();
            var checkboxes = document.querySelectorAll('input[type="checkbox"]');
            if (!checkboxes.length) return;

            var checkedCount = 0;
            for (var i = 0; i < checkboxes.length; i++) {
                if (state[checkboxes[i].id]) {
                    checkedCount++;
                }
            }

            var percent = Math.round((checkedCount / checkboxes.length) * 100);
            document.getElementById('overall-progress').innerText = percent + '%';
        }

        function buildCalendar() {
            var container = document.getElementById('calendar');
            container.innerHTML = '';
            var state = getSavedState();

            var dayNum = 1;

            for (var week = 1; week <= 4; week++) {
                var weekBlock = document.createElement('div');
                weekBlock.className = 'week-block week-' + week;

                var isRetWeek = week >= 3;
                var badgeLabel = week === 1 ? 'Адаптация' : week === 2 ? 'Базовый уход' : week === 3 ? 'Ретинол (2 р/нед)' : 'Ретинол (3 р/нед)';

                var weekHeader = document.createElement('div');
                weekHeader.className = 'week-title';
                weekHeader.innerHTML = 'Неделя ' + week + ' <span class="badge ' + (isRetWeek ? 'retinol' : '') + '">' + badgeLabel + '</span>';
                weekBlock.appendChild(weekHeader);

                var grid = document.createElement('div');
                grid.className = 'days-grid';

                for (var dayIdx = 0; dayIdx < 7; dayIdx++) {
                    var retinol = isRetinolDay(week, dayIdx);
                    var card = document.createElement('div');
                    card.className = 'day-card' + (retinol ? ' has-retinol' : '');

                    var m1 = 'd' + dayNum + '_m1';
                    var m2 = 'd' + dayNum + '_m2';
                    var m3 = 'd' + dayNum + '_m3';

                    var e1 = 'd' + dayNum + '_e1';
                    var e2 = 'd' + dayNum + '_e2';
                    var e3 = 'd' + dayNum + '_e3';

                    var eveningContent = '';
                    if (retinol) {
                        eveningContent = 
                            '<label class="task-item">' +
                                '<input type="checkbox" id="' + e1 + '" ' + (state[e1] ? 'checked' : '') + ' onchange="toggleTask(\'' + e1 + '\')">' +
                                '<span>Умывание муссом с алоэ</span>' +
                            '</label>' +
                            '<label class="task-item">' +
                                '<input type="checkbox" id="' + e2 + '" ' + (state[e2] ? 'checked' : '') + ' onchange="toggleTask(\'' + e2 + '\')">' +
                                '<span class="retinol-highlight">Сухая кожа (10 мин) + Retiderm 0.25</span>' +
                            '</label>' +
                            '<label class="task-item">' +
                                '<input type="checkbox" id="' + e3 + '" ' + (state[e3] ? 'checked' : '') + ' onchange="toggleTask(\'' + e3 + '\')">' +
                                '<span>Пауза 20 мин + крем NMF-Protection</span>' +
                            '</label>';
                    } else {
                        eveningContent = 
                            '<label class="task-item">' +
                                '<input type="checkbox" id="' + e1 + '" ' + (state[e1] ? 'checked' : '') + ' onchange="toggleTask(\'' + e1 + '\')">' +
                                '<span>Умывание муссом с алоэ</span>' +
                            '</label>' +
                            '<label class="task-item">' +
                                '<input type="checkbox" id="' + e2 + '" ' + (state[e2] ? 'checked' : '') + ' onchange="toggleTask(\'' + e2 + '\')">' +
                                '<span>Крем NMF-Protection</span>' +
                            '</label>';
                    }

                    card.innerHTML = 
                        '<div class="day-header">' +
                            '<span class="day-name">' + daysOfWeek[dayIdx] + '</span>' +
                            '<span class="day-num">День ' + dayNum + '</span>' +
                        '</div>' +
                        '<div class="section-box">' +
                            '<span class="section-time">☀️ Утро (10:00)</span>' +
                            '<label class="task-item">' +
                                '<input type="checkbox" id="' + m1 + '" ' + (state[m1] ? 'checked' : '') + ' onchange="toggleTask(\'' + m1 + '\')">' +
                                '<span>Умывание муссом с алоэ</span>' +
                            '</label>' +
                            '<label class="task-item">' +
                                '<input type="checkbox" id="' + m2 + '" ' + (state[m2] ? 'checked' : '') + ' onchange="toggleTask(\'' + m2 + '\')">' +
                                '<span>Крем NMF-Protection</span>' +
                            '</label>' +
                            '<label class="task-item">' +
                                '<input type="checkbox" id="' + m3 + '" ' + (state[m3] ? 'checked' : '') + ' onchange="toggleTask(\'' + m3 + '\')">' +
                                '<span>Солнцезащитный крем (SPF)</span>' +
                            '</label>' +
                        '</div>' +
                        '<div class="section-box">' +
                            '<span class="section-time">🌙 Вечер (21:00)</span>' +
                            eveningContent +
                        '</div>';

                    grid.appendChild(card);
                    dayNum++;
                }

                weekBlock.appendChild(grid);
                container.appendChild(weekBlock);
            }

            updateProgress();
        }

        function filterWeek(weekNum, btn) {
            var tabs = document.querySelectorAll('.tab-btn');
            for (var i = 0; i < tabs.length; i++) {
                tabs[i].classList.remove('active');
            }
            if (btn) btn.classList.add('active');

            var blocks = document.querySelectorAll('.week-block');
            for (var j = 0; j < blocks.length; j++) {
                if (weekNum === 'all') {
                    blocks[j].style.display = 'block';
                } else {
                    if (blocks[j].classList.contains('week-' + weekNum)) {
                        blocks[j].style.display = 'block';
                    } else {
                        blocks[j].style.display = 'none';
                    }
                }
            }
        }

        function resetProgress() {
            if (confirm('Сбросить отмеченный прогресс?')) {
                localStorage.removeItem('skincare_gh_pages_data');
                buildCalendar();
            }
        }

        window.onload = function() {
            buildCalendar();
        };
    </script>
